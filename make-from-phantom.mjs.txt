import fs from 'fs'
import bs58 from 'bs58'

const base58Key = '4fVQZL4NJ7xG8r1CiU3ULPMkn6RmaRPeBYiZ9rJ7SwSzgQMKAtUtEJtmxokaHxpE3GqNVrAhV4Td4fVBsTvLpddE'

const secret = bs58.decode(base58Key)
fs.writeFileSync('./wallet.json', JSON.stringify(Array.from(secret)))
console.log('DONE! wallet.json created')

import { createUmi } from '@metaplex-foundation/umi-bundle-defaults'
const umi = createUmi('https://api.mainnet-beta.solana.com')
const kp = umi.eddsa.createKeypairFromSecretKey(new Uint8Array(secret))
console.log('Public Key:', kp.publicKey)
